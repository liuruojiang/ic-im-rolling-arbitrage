def _next_session_or_record(day: date) -> date:
    """Strict next session. While recording, a recorded calendar gap continues
    on the nominal weekday date only to keep acquiring evidence.

    _require_official_exchange_calendar has already appended the failure to the
    diagnostic context, so build_live_trade_signal converts the final result to
    diagnostic-only; the nominal date can never become a formal target.
    """
    try:
        return _roll_forward_exchange_day(day, require_official=True)
    except CalendarCoverageError:
        if not _recording_calendar_gap():
            raise
        return _roll_forward_exchange_day(day)
